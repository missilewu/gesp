# 数据处理+排序

```cpp
#include <bits/stdc++.h>
using namespace std;

// 定义商品结构体，封装属性
struct Item {
    string name;
    int price;
    int priority;
};

// 修正：将cmp函数移到结构体外部，符合C++语法
bool cmp(const Item &a, const Item &b) {
    // 第一层：优先级升序
    if (a.priority != b.priority) 
        return a.priority < b.priority;
    // 第二层：价格升序
    if (a.price != b.price) 
        return a.price < b.price;
    // 第三层：名字典序升序
    return a.name < b.name;
}

Item items[1005]; // 定义存储商品的数组
string s[1005]; // 存储购买的商品名

int main(){
    int M, N;
    cin >> M >> N;


    // 输入N个商品的属性
    for (int i = 0; i < N; ++i) {
        cin >> items[i].name >> items[i].price >> items[i].priority;
    }

    // 第一次排序：按购买优先级排序
    sort(items, items + N, cmp);

    int budget = M;
    int count = 0; // 记录购买的商品数量

    // 贪心购买
    for (int i = 0; i < N; ++i) {
        if (items[i].price <= budget) {
            budget -= items[i].price;
            s[count++] = items[i].name; // 将商品名存入字符串数组
        }
    }

    // 第二次排序：按字典序升序排序
    sort(s, s + count);

    // 输出购买的商品名
    for (int i = 0; i < count; ++i) {
        cout << s[i] << endl;
    }

    return 0;
}
```

# 二维数组和矩阵的遍历

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1010; // 定义数组最大尺寸，适配M,N≤10³
int a[MAXN][MAXN];     // 存储地形图的海拔高度

int main() {
    ios::sync_with_stdio(false); // 加速输入输出，避免大数据下超时
    cin.tie(0);

    int M, N, H;
    cin >> M >> N >> H;

    // 输入M行N列的数据
    for (int i = 1; i <= M; ++i) {
        for (int j = 1; j <= N; ++j) {
            cin >> a[i][j];
        }
    }

    int max_sum = 0; // 记录满足条件的最大和，初始化为0（题目保证有解）

    // 枚举所有3×3子矩阵的左上角(i,j)，确保不越界
    for (int i = 1; i <= M - 2; ++i) {
        for (int j = 1; j <= N - 2; ++j) {
            int local_max = a[i][j]; // 当前子矩阵的最大高度，初始化为第一个元素
            int local_min = a[i][j]; // 当前子矩阵的最小高度，初始化为第一个元素
            int local_sum = 0;       // 当前子矩阵的高度和，初始化为0

            // 遍历3×3的所有元素
            for (int x = 0; x < 3; ++x) { // x为行偏移：0,1,2
                for (int y = 0; y < 3; ++y) { // y为列偏移：0,1,2
                    int h = a[i + x][j + y]; // 当前元素的高度
                    local_sum += h;          // 累加和

                    // 更新最大值
                    if (h > local_max) {
                        local_max = h;
                    }
                    // 更新最小值
                    if (h < local_min) {
                        local_min = h;
                    }
                }
            }

            // 判断是否满足建造条件，满足则更新最大和
            if (local_max - local_min <= H) {
                if (local_sum > max_sum) {
                    max_sum = local_sum;
                }
            }
        }
    }

    // 输出结果
    cout << max_sum << endl;
    return 0;
}
```
