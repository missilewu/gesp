
```C++
#include <iostream>
using namespace std;

int main() {
    // 1. 输入处理：读取n和m
    int n, m;
    cin >> n >> m;

    // 2. 定义标记数组，初始化为false（未到达）
    // GESP中常用bool数组做标记，长度n对应0~n-1号同学
    bool arrived[1001] = {false};  // 题目约定n≤1000，数组开1001足够
    for(int i = 0; i < n; i++) {
    	arrived[i] = false;
	}

    // 3. 读取m次报数，标记已到达的同学
    for (int i = 0; i < m; i++) {
        int num;
        cin >> num;
        arrived[num] = true;  // 重复报数也只会设为true，自动去重
    }

    // 4. 收集未到达的同学编号
    bool all = true;
    for (int i = 0; i < n; i++) {
    	if(!arrived[i]) {
    		cout << i << " ";
    		all = false;
		}
	}
	
	if(all) {
		cout << n; 
	}
	
	cout << endl;

    return 0;
}
```
