以下是符合C风格（使用`struct`和原始指针，而非C++类或STL）的单循环链表标准定义及核心操作代码。

### 1. 单循环链表的定义

在单循环链表中，最后一个节点的 `next` 指针不指向 `NULL`，而是指向头节点（或首元节点），形成一个环。

为了操作方便，通常有两种实现方式：

1. **带头节点（Head Node）**：有一个不存数据的头节点，`head->next` 指向第一个实际数据节点。空表时 `head->next == head`。
2. **不带头节点**：直接指向第一个数据节点。空表时指针为 `NULL`。

**在竞赛中，为了减少边界判断（如空表插入、删除第一个元素），推荐使用“带头节点”的方式，或者维护一个指向尾节点的指针。** 下面的代码采用**带头节点**的标准实现，这样逻辑更统一，不易出错。

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// 定义节点结构体
typedef struct Node {
    int data;               // 数据域
    struct Node* next;      // 指针域
} Node;

// 定义链表结构体（可选，也可以直接用 Node* 表示链表）
typedef struct {
    Node* head;             // 指向头节点（不存有效数据）
    int size;               // 记录链表长度，方便操作
} CircularLinkedList;

// 初始化链表
void initList(CircularLinkedList* list) {
    list->head = (Node*)malloc(sizeof(Node));
    if (!list->head) {
        printf("Memory allocation failed!\n");
        exit(1);
    }
    list->head->data = 0;   // 头节点数据无意义
    list->head->next = list->head; // 关键：指向自己，形成环
    list->size = 0;
}

// 创建新节点
Node* createNode(int value) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (!newNode) {
        printf("Memory allocation failed!\n");
        exit(1);
    }
    newNode->data = value;
    newNode->next = NULL;
    return newNode;
}
```

### 2. 核心操作实现

#### A. 遍历 (Traversal)

由于是循环链表，终止条件不是 `p != NULL`，而是 `p != head`。

```c
void traverseList(CircularLinkedList* list) {
    if (list->size == 0) {
        printf("List is empty.\n");
        return;
    }

    Node* p = list->head->next; // 从第一个实际节点开始
    printf("List elements: ");
    do {
        printf("%d -> ", p->data);
        p = p->next;
    } while (p != list->head); // 回到头节点则停止
    printf("(Head)\n");
}
```

#### B. 搜索 (Search)

查找值为 `val` 的节点，返回该节点指针，若未找到返回 `NULL`。

```c
Node* searchList(CircularLinkedList* list, int val) {
    if (list->size == 0) return NULL;

    Node* p = list->head->next;
    do {
        if (p->data == val) {
            return p;
        }
        p = p->next;
    } while (p != list->head);

    return NULL; // 未找到
}
```

#### C. 插入 (Insertion)

这里演示**头插法**（在第一个数据节点前插入）和**尾插法**。  
_注意：在循环链表中，插入操作需要小心维护环的闭合。_

**尾插法（推荐，常用于构建队列或约瑟夫环）：**

```c
void insertTail(CircularLinkedList* list, int val) {
    Node* newNode = createNode(val);
    
    // 找到尾节点：即 head 的前一个节点
    // 因为有 head 节点，tail->next == head
    Node* tail = list->head;
    while (tail->next != list->head) {
        tail = tail->next;
    }

    // 插入操作
    newNode->next = list->head; // 新节点指向头
    tail->next = newNode;       // 原尾节点指向新节点
    
    list->size++;
}
```

**指定位置插入（在第 pos 个位置后插入，pos从0开始计数，0表示插在头节点后）：**

```c
bool insertAt(CircularLinkedList* list, int pos, int val) {
    if (pos < 0 || pos > list->size) return false;

    Node* p = list->head;
    // 移动到第 pos 个节点（即插入位置的前驱）
    for (int i = 0; i < pos; i++) {
        p = p->next;
    }

    Node* newNode = createNode(val);
    newNode->next = p->next;
    p->next = newNode;
    
    list->size++;
    return true;
}
```

#### D. 删除 (Deletion)

删除值为 `val` 的第一个节点。需要找到待删除节点的前驱节点。

```c
bool deleteNode(CircularLinkedList* list, int val) {
    if (list->size == 0) return false;

    Node* prev = list->head;
    Node* curr = list->head->next;

    // 遍历寻找
    while (curr != list->head) {
        if (curr->data == val) {
            // 找到目标，执行删除
            prev->next = curr->next;
            free(curr);
            list->size--;
            return true;
        }
        prev = curr;
        curr = curr->next;
    }

    return false; // 未找到
}
```

#### E. 反转 (Reverse)

单循环链表的反转与单链表类似，但最后需要将新的尾节点（原头节点的后继）重新连回头节点。  
**注意**：对于带头节点的循环链表，反转的是数据节点部分，头节点 `head` 保持不变，始终作为入口。

```c
void reverseList(CircularLinkedList* list) {
    if (list->size <= 1) return;

    Node* prev = NULL;
    Node* curr = list->head->next; // 第一个数据节点
    Node* nextNode = NULL;
    Node* firstDataNode = curr;    // 记录原来的第一个数据节点，它将成为新的尾节点

    // 标准单链表反转逻辑
    while (curr != list->head) {
        nextNode = curr->next;
        curr->next = prev;
        prev = curr;
        curr = nextNode;
    }

    // 此时：
    // prev 指向原来的最后一个节点（现在是新的第一个数据节点）
    // firstDataNode 指向原来的第一个节点（现在是新的最后一个数据节点）
    
    // 重新建立循环连接
    list->head->next = prev;       // 头节点指向新的首元节点
    firstDataNode->next = list->head; // 新的尾节点指向头节点
}
```

### 3. 完整测试样例 (Main Function)

将上述代码组合在一起进行测试：

```c
int main() {
    CircularLinkedList list;
    initList(&list);

    printf("--- Inserting elements (Tail): 10, 20, 30 ---\n");
    insertTail(&list, 10);
    insertTail(&list, 20);
    insertTail(&list, 30);
    traverseList(&list);

    printf("\n--- Inserting 5 at position 0 (Head) ---\n");
    insertAt(&list, 0, 5);
    traverseList(&list);

    printf("\n--- Searching for 20 ---\n");
    Node* found = searchList(&list, 20);
    if (found) printf("Found: %d\n", found->data);
    else printf("Not found.\n");

    printf("\n--- Deleting 20 ---\n");
    if (deleteNode(&list, 20)) {
        printf("Deleted successfully.\n");
        traverseList(&list);
    } else {
        printf("Delete failed.\n");
    }

    printf("\n--- Reversing the list ---\n");
    reverseList(&list);
    traverseList(&list);

    // 清理内存（竞赛中通常由系统回收，但良好习惯需释放）
    Node* p = list.head->next;
    while (p != list.head) {
        Node* temp = p;
        p = p->next;
        free(temp);
    }
    free(list.head);

    return 0;
}
```

### 4. 竞赛中的关键点提示 (GESP/NOI/IOI)

1. **空表判断**：在循环链表中，`list->head->next == list->head` 是判断空表的唯一标准。
2. **死循环风险**：遍历时的 `do-while` 或 `while` 条件必须严格检查是否回到了 `head`，否则会导致无限循环（TLE）。

3. **内存泄漏**：虽然在线评测系统（OJ）在程序结束后会回收内存，但在删除节点时务必 `free()`，养成良好的C语言编程习惯。
4. **指针操作顺序**：在插入和删除时，一定要先连接新指针，再断开旧指针，防止链表断裂丢失后续节点。

这段代码完全遵循C风格，没有使用C++的类封装，非常适合用于理解底层指针操作和应对相关考试。
