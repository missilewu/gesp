在NOI（全国青少年信息学奥林匹克竞赛）和IOI（国际信息学奥林匹克竞赛）以及GESP（CCF编程能力等级认证）的C++考试中，虽然`std::list`或`std::vector`在实际工程中更常用，但**手写链表**是考察指针操作、内存管理和算法逻辑的核心考点。

特别是**双向循环链表**（Doubly Circular Linked List），由于其头尾相连且双向可遍历的特性，在处理约瑟夫环、滑块谜题或需要频繁首尾插入删除的题目时非常高效。

以下是采用**C风格（C-style）**实现的完整代码，包含定义、初始化、遍历、搜索、插入、删除和反转操作。代码注重指针操作的细节，符合竞赛中对底层逻辑的考察要求。

### 1. 结构体定义与节点创建

双向循环链表的核心在于：

- 每个节点有 `prev` (前驱) 和 `next` (后继)。
- 头节点的 `prev` 指向尾节点。
- 尾节点的 `next` 指向头节点。
- 空表时，头节点的 `next` 和 `prev` 均指向自己。

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// 定义节点结构体
typedef struct Node {
    int data;
    struct Node* prev;
    struct Node* next;
} Node;

// 定义链表结构体（通常只需一个头指针即可，但为了封装性，这里定义一个结构体）
typedef struct DoublyCircularList {
    Node* head;
    int size;
} DoublyCircularList;

// 创建新节点
Node* createNode(int value) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (!newNode) {
        printf("Memory allocation failed!\n");
        exit(1);
    }
    newNode->data = value;
    newNode->prev = NULL;
    newNode->next = NULL;
    return newNode;
}

// 初始化链表
void initList(DoublyCircularList* list) {
    list->head = createNode(0); // 创建一个哨兵头节点（dummy head），数据域通常不使用或存大小
    list->head->next = list->head;
    list->head->prev = list->head;
    list->size = 0;
}

// 检查链表是否为空
bool isEmpty(DoublyCircularList* list) {
    return list->size == 0;
}
```

### 2. 尾部插入（Append）与头部插入（Prepend）

在循环链表中，利用哨兵节点可以简化边界判断。

```c
// 在链表尾部插入元素
void append(DoublyCircularList* list, int value) {
    Node* newNode = createNode(value);
    Node* tail = list->head->prev; // 当前尾节点

    // 连接新节点
    newNode->next = list->head;
    newNode->prev = tail;
    
    // 更新原尾节点和头节点
    tail->next = newNode;
    list->head->prev = newNode;

    list->size++;
}

// 在链表头部插入元素（头节点之后）
void prepend(DoublyCircularList* list, int value) {
    Node* newNode = createNode(value);
    Node* first = list->head->next; // 当前第一个实际节点

    // 连接新节点
    newNode->next = first;
    newNode->prev = list->head;

    // 更新原首节点和头节点
    first->prev = newNode;
    list->head->next = newNode;

    list->size++;
}
```

### 3. 遍历操作（Traversal）

遍历时需注意终止条件：当指针回到 `head` 时停止。

```c
// 正向遍历打印
void traverseForward(DoublyCircularList* list) {
    if (isEmpty(list)) {
        printf("List is empty.\n");
        return;
    }
    Node* curr = list->head->next;
    printf("Forward: ");
    while (curr != list->head) {
        printf("%d ", curr->data);
        curr = curr->next;
    }
    printf("\n");
}

// 反向遍历打印
void traverseBackward(DoublyCircularList* list) {
    if (isEmpty(list)) {
        printf("List is empty.\n");
        return;
    }
    Node* curr = list->head->prev;
    printf("Backward: ");
    while (curr != list->head) {
        printf("%d ", curr->data);
        curr = curr->prev;
    }
    printf("\n");
}
```

### 4. 搜索操作（Search）

返回找到的节点指针，若未找到返回 `NULL`。

```c
Node* search(DoublyCircularList* list, int value) {
    if (isEmpty(list)) return NULL;
    
    Node* curr = list->head->next;
    while (curr != list->head) {
        if (curr->data == value) {
            return curr;
        }
        curr = curr->next;
    }
    return NULL;
}
```

### 5. 插入指定位置（Insert After）

在竞赛中，通常是在某个已知节点后插入。

```c
// 在指定节点 target 之后插入新值
// 如果 target 为 NULL 或不在链表中，操作失败
bool insertAfter(DoublyCircularList* list, Node* target, int value) {
    if (target == NULL) return false;
    
    // 简单校验 target 是否属于该链表（可选，竞赛中通常假设输入合法）
    // 这里为了效率省略校验
    
    Node* newNode = createNode(value);
    Node* nextNode = target->next;

    newNode->prev = target;
    newNode->next = nextNode;

    target->next = newNode;
    nextNode->prev = newNode;

    list->size++;
    return true;
}
```

### 6. 删除操作（Delete）

删除节点时需要小心处理前后指针的连接，并释放内存。

```c
// 删除指定节点
// 返回 true 表示删除成功
bool deleteNode(DoublyCircularList* list, Node* target) {
    if (target == NULL || target == list->head) return false; // 不能删除哨兵头节点

    Node* prevNode = target->prev;
    Node* nextNode = target->next;

    // 重新连接
    prevNode->next = nextNode;
    nextNode->prev = prevNode;

    free(target);
    list->size--;
    return true;
}

// 按值删除（删除第一个匹配的节点）
bool deleteByValue(DoublyCircularList* list, int value) {
    Node* target = search(list, value);
    return deleteNode(list, target);
}
```

### 7. 链表反转（Reverse）

这是高频考点。对于双向链表，反转只需要交换每个节点的 `prev` 和 `next` 指针，最后还需要交换头节点的指向（或者简单地交换遍历方向，但在物理结构上通常要求彻底反转）。

**注意**：在双向循环链表中，如果仅仅交换每个节点的 `prev` 和 `next`，链表的逻辑方向就反了。由于是循环的，原来的 `head->next` 变成了 `head->prev`。为了保持 `head` 依然作为入口且 `head->next` 指向新的第一个元素，我们需要做特殊处理，或者直接交换整个链表的 `next` 和 `prev` 逻辑（即遍历时走 `prev` 指针）。

下面给出**物理结构完全反转**的实现：交换所有节点的指针，并调整头节点的连接关系。

```c
void reverseList(DoublyCircularList* list) {
    if (isEmpty(list) || list->size == 1) return;

    Node* curr = list->head->next;
    Node* temp;

    // 1. 交换每个节点的 prev 和 next 指针
    // 注意：需要遍历整个链表一圈
    do {
        temp = curr->next;
        curr->next = curr->prev;
        curr->prev = temp;
        
        curr = temp; // 移动到下一个（原本的下一个，现在已经是 prev 了，所以用 temp 保存的原 next）
    } while (curr != list->head);

    // 2. 此时，逻辑上的“下一个”节点其实是在物理上的 prev 方向
    // 为了保持 head->next 指向新的首元素（原尾元素），我们需要交换 head 的 next 和 prev
    temp = list->head->next;
    list->head->next = list->head->prev;
    list->head->prev = temp;
}
```

### 8. 主函数测试样例

```c
int main() {
    DoublyCircularList list;
    initList(&list);

    printf("--- Inserting elements ---\n");
    append(&list, 10);
    append(&list, 20);
    append(&list, 30);
    prepend(&list, 5);
    traverseForward(&list);   // 输出: 5 10 20 30
    traverseBackward(&list);  // 输出: 30 20 10 5

    printf("\n--- Searching ---\n");
    Node* found = search(&list, 20);
    if (found) {
        printf("Found node with value: %d\n", found->data);
        // 在 20 后面插入 25
        insertAfter(&list, found, 25);
    }
    traverseForward(&list);   // 输出: 5 10 20 25 30

    printf("\n--- Deleting ---\n");
    deleteByValue(&list, 5);  // 删除头
    deleteByValue(&list, 30); // 删除尾
    traverseForward(&list);   // 输出: 10 20 25

    printf("\n--- Reversing ---\n");
    reverseList(&list);
    traverseForward(&list);   // 输出: 25 20 10 (顺序反转)
    traverseBackward(&list);  // 输出: 10 20 25

    // 清理内存（竞赛中通常由系统回收，但好习惯是手动释放）
    while (!isEmpty(&list)) {
        deleteNode(&list, list.head->next);
    }
    free(list.head);

    return 0;
}
```

### 竞赛中的关键点提示（GESP/NOI/IOI）

1. **哨兵节点（Dummy Head）**：代码中使用了 `head` 作为哨兵节点，不存储有效数据。这极大地简化了在第一位置插入和删除第一位置元素的代码逻辑，避免了大量的 `if (head == NULL)` 判断。
2. **循环条件**：遍历时的核心条件是 `curr != list->head`。如果是空表，`head->next` 就是 `head`，循环一次都不会执行，逻辑自洽。
3. **指针交换顺序**：在插入和删除操作中，必须严格按照顺序修改指针，防止链表断裂丢失节点。例如删除时，先保存 `prev` 和 `next`，再重连，最后 `free`。
4. **内存泄漏**：在C风格代码中，`malloc` 对应的 `free` 必须由程序员管理。在竞赛中，如果程序运行时间短，操作系统会回收；但在长运行时间题目或多次测试用例中，忘记 `free` 可能导致 MLE (Memory Limit Exceeded)。
5. **反转的陷阱**：双向链表反转不仅仅是单向链表那样改变 `next` 指针，必须同时交换 `prev` 和 `next`，并且要注意循环方向的维持。
