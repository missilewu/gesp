虽然现代C++推荐使用`std::list`或`std::vector`，但在竞赛（尤其是NOI/IOI早期题目或考察底层能力的GESP高年级考试）中，往往要求选手能够**手写链表**，以便更好地理解指针操作、内存分配以及在特定场景下（如静态链表优化）的性能控制。

以下是基于 **C风格（使用结构体和原始指针）** 的单链表标准定义及核心操作样例代码。代码注重规范性、边界条件处理（如空表、头节点操作），符合竞赛编码习惯。

### 1. 单链表的定义

在C/C++竞赛风格中，通常使用 `struct` 定义节点，并使用 `typedef` 简化类型名称。

```cpp
#include <iostream>
#include <cstdlib> // 用于 malloc, free 或 new, delete (这里演示使用 new/delete 以符合C++习惯，但风格是C式的结构体操作)

using namespace std;

// 定义链表节点结构体
struct Node {
    int data;       // 数据域，根据题目要求可改为 long long, char 等
    Node* next;     // 指针域，指向下一个节点

    // 构造函数（可选，方便初始化）
    Node(int val) : data(val), next(NULL) {}
};

// 定义链表头指针类型（可选，方便书写）
typedef Node* LinkedList;
```

### 2. 核心操作样例代码

以下代码包含：创建（头插/尾插）、遍历、搜索、插入、删除、反转。为了方便演示，所有操作均封装为函数，并在 `main` 函数中提供测试用例。

#### 完整代码实现

```cpp
#include <iostream>
#include <cassert>

using namespace std;

// --- 1. 定义结构体 ---
struct Node {
    int data;
    Node* next;
    Node(int val) : data(val), next(NULL) {}
};

// --- 2. 遍历 (Traversal) ---
// 时间复杂度: O(n)
void traverse(LinkedList head) {
    Node* curr = head;
    if (curr == NULL) {
        cout << "List is empty." << endl;
        return;
    }
    cout << "List: ";
    while (curr != NULL) {
        cout << curr->data << " -> ";
        curr = curr->next;
    }
    cout << "NULL" << endl;
}

// --- 3. 搜索 (Search) ---
// 返回值为找到的节点指针，若未找到返回 NULL
// 时间复杂度: O(n)
Node* search(LinkedList head, int target) {
    Node* curr = head;
    while (curr != NULL) {
        if (curr->data == target) {
            return curr;
        }
        curr = curr->next;
    }
    return NULL;
}

// --- 4. 头部插入 (Insert at Head) ---
// 常用于快速建表，时间复杂度: O(1)
// 注意：需要修改头指针，所以传入头指针的引用 (Node*& ) 或二级指针
void insertAtHead(LinkedList& head, int val) {
    Node* newNode = new Node(val);
    newNode->next = head;
    head = newNode;
}

// --- 5. 尾部插入 (Insert at Tail) ---
// 时间复杂度: O(n)，若维护尾指针可优化至 O(1)
void insertAtTail(LinkedList& head, int val) {
    Node* newNode = new Node(val);
    if (head == NULL) {
        head = newNode;
        return;
    }
    Node* curr = head;
    while (curr->next != NULL) {
        curr = curr->next;
    }
    curr->next = newNode;
}

// --- 6. 指定位置插入 (Insert after a specific node) ---
// 在节点 prevNode 之后插入新节点
// 时间复杂度: O(1) (前提是已经找到了 prevNode)
void insertAfter(Node* prevNode, int val) {
    if (prevNode == NULL) {
        cout << "Error: Previous node cannot be NULL." << endl;
        return;
    }
    Node* newNode = new Node(val);
    newNode->next = prevNode->next;
    prevNode->next = newNode;
}

// --- 7. 删除节点 (Delete by value) ---
// 删除第一个值为 target 的节点
// 时间复杂度: O(n)
// 需要处理删除头节点的特殊情况
bool deleteNode(LinkedList& head, int target) {
    if (head == NULL) return false;

    // 特殊情况：删除头节点
    if (head->data == target) {
        Node* temp = head;
        head = head->next;
        delete temp; // 释放内存
        return true;
    }

    // 一般情况：寻找目标节点的前驱节点
    Node* curr = head;
    while (curr->next != NULL && curr->next->data != target) {
        curr = curr->next;
    }

    // 如果找到了目标节点 (curr->next 就是目标)
    if (curr->next != NULL) {
        Node* temp = curr->next;
        curr->next = curr->next->next; // 跳过目标节点
        delete temp;
        return true;
    }

    return false; // 未找到
}

// --- 8. 反转链表 (Reverse) ---
// 迭代法实现，三个指针：prev, curr, nextTemp
// 时间复杂度: O(n), 空间复杂度: O(1)
void reverseList(LinkedList& head) {
    Node* prev = NULL;
    Node* curr = head;
    Node* nextTemp = NULL;

    while (curr != NULL) {
        nextTemp = curr->next; // 暂存下一个节点
        curr->next = prev;     // 当前节点指向前一个节点（反转）
        prev = curr;           // prev 前移
        curr = nextTemp;       // curr 前移
    }
    head = prev; // 更新头指针为新的头（原尾节点）
}

// --- 9. 释放链表内存 (Clear) ---
// 竞赛中通常程序结束系统会自动回收，但良好的习惯是手动释放
void clearList(LinkedList& head) {
    Node* curr = head;
    while (curr != NULL) {
        Node* temp = curr;
        curr = curr->next;
        delete temp;
    }
    head = NULL;
}

// ================= 主函数测试 =================
int main() {
    LinkedList myList = NULL; // 初始化空链表

    cout << "--- 1. 头部插入测试 (5, 4, 3) ---" << endl;
    insertAtHead(myList, 3);
    insertAtHead(myList, 4);
    insertAtHead(myList, 5);
    traverse(myList); // 预期: 5 -> 4 -> 3

    cout << "\n--- 2. 尾部插入测试 (追加 6, 7) ---" << endl;
    insertAtTail(myList, 6);
    insertAtTail(myList, 7);
    traverse(myList); // 预期: 5 -> 4 -> 3 -> 6 -> 7

    cout << "\n--- 3. 搜索测试 (查找 4 和 9) ---" << endl;
    Node* found = search(myList, 4);
    if (found) cout << "Found 4: " << found->data << endl;
    else cout << "4 not found." << endl;

    found = search(myList, 9);
    if (found) cout << "Found 9: " << found->data << endl;
    else cout << "9 not found." << endl;

    cout << "\n--- 4. 插入测试 (在 4 后面插入 100) ---" << endl;
    Node* node4 = search(myList, 4);
    if (node4) {
        insertAfter(node4, 100);
        traverse(myList); // 预期: ... 4 -> 100 -> 3 ...
    }

    cout << "\n--- 5. 删除测试 (删除头节点 5, 删除中间节点 100, 删除不存在的 9) ---" << endl;
    deleteNode(myList, 5);
    traverse(myList);
    deleteNode(myList, 100);
    traverse(myList);
    if (!deleteNode(myList, 9)) {
        cout << "Failed to delete 9 (not found)." << endl;
    }

    cout << "\n--- 6. 反转测试 ---" << endl;
    traverse(myList); // 当前状态
    reverseList(myList);
    traverse(myList); // 预期完全倒序

    cout << "\n--- 7. 清理内存 ---" << endl;
    clearList(myList);
    traverse(myList); // 预期: List is empty.

    return 0;
}
```

### 关键点解析 (针对GESP/NOI/IOI)

1. **指针引用的使用 (`Node*& head`)**：
    
    - 在C++中，如果函数内部需要修改指针本身（例如改变头指针 `head` 指向新的节点，或者将 `head` 置为 `NULL`），必须传递指针的**引用**（`&`）或者使用**二级指针**（`Node**`）。
    - 在上述代码的 `insertAtHead`, `insertAtTail`, `deleteNode`, `reverseList` 中，都使用了 `LinkedList& head`，这是C++写法中最简洁且安全的方式。如果是纯C语言，则需要写成 `void insertAtHead(Node** head)`。
2. **边界条件处理**：
    
    - **空表检查**：几乎所有操作开始前都要检查 `head == NULL`。
    - **头节点删除**：删除操作最易错的地方是删除第一个元素。代码中专门用 `if (head->data == target)` 处理了这种情况，避免访问空指针的 `next`。
    - **前驱节点**：单向链表无法直接获取前驱节点，因此删除或插入到某节点前时，通常需要从头遍历找到 `prev` 节点（即 `curr->next == target` 时的 `curr`）。
3. **内存管理**：
    
    - 竞赛中虽然程序结束后OS会回收内存，但在编写涉及大量动态分配的代码（如大数据量模拟）时，及时 `delete` 不再使用的节点可以防止内存超限（MLE）。
    - 注意：在GESP高年级考试中，可能会考察手动管理内存的意识。
4. **反转算法 (Iterative Reverse)**：
    
    - 这是高频考题。核心思想是维护三个指针：`prev` (前驱), `curr` (当前), `nextTemp` (后继暂存)。
    - 循环体内顺序不能乱：先存后继，再断连反转，最后整体后移。
5. **静态链表 (Static Linked List)**：
    
    - **特别提示**：在NOI/IOI的高级组别中，有时会考察**静态链表**（使用数组模拟链表）。这是因为 `new` 操作在极端数据量下较慢，且容易产生内存碎片。
    - 静态链表定义示例：
    
    ```cpp
    const int MAXN = 100005;
    struct Node {
        int data;
        int next; // 存放下标索引，而不是指针
    } pool[MAXN];
    int head, avail; // head是头节点下标，avail是空闲池指针
    ```
    
    - 如果你参加的是GESP五级或NOI普及组，掌握上述动态指针链表即可；若冲刺提高组或省选，建议同时掌握数组模拟链表。
